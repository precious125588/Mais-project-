STEP 5 — Welcome/goodbye Unicode names (𝑷𝑹𝑬𝑪𝑰𝑶𝑼𝑺 x boxes fix)
  Root cause: welcomeCards.cjs only registers a FIXED list of font paths;
  on Railway/Alpine those exact paths are missing so it falls back to Jimp
  bitmap fonts (Basic Latin only → □ for mathematical letters U+1D400+).
  In mias/lib/welcomeCards.cjs replace the font-registration block with a
  directory walk that registers EVERY font found:

      const fs=require('fs'), path=require('path');
      function walk(dir, depth){
        if (depth > 6) return;
        let ents=[]; try{ ents=fs.readdirSync(dir,{withFileTypes:true}); }catch{return;}
        for (const e of ents){
          const p=path.join(dir, e.name);
          if (e.isDirectory()) walk(p, depth+1);
          else if (/\.(ttf|ttc|otf)$/i.test(e.name)){ try{ GlobalFonts.registerFromPath(p); }catch{} }
        }
      }
      for (const root of ['/usr/share/fonts','/usr/local/share/fonts','/root/.fonts']) walk(root, 0);

  AND make sure your Dockerfile installs the Noto Math + Symbols fonts:
      RUN apk add --no-cache font-noto font-noto-math font-noto-symbols font-noto-emoji ttf-dejavu
  (Debian base: apt-get install -y fonts-noto fonts-noto-math fonts-dejavu)
